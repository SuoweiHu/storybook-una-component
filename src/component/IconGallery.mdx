import { Meta, IconGallery, IconItem } from '@storybook/addon-docs/blocks';
import { Icon } from './Icon';

Every icon in every variant (white icons are shown on a dark swatch): {socialIconNames.length + utilityIconNames.length + heroIconNames.length} icons × {iconVariants.length} variants. See [Icon docs](?path=/docs/icon--docs) for usage.

{iconVariants.map((variant) => (
  <div key={variant}>

      ))}
    </IconGallery>
    <h3>Small use – {variant}</h3>
    <IconGallery>

        </IconItem>
      ))}
    </IconGallery>
    {heroIconSets.map((set) => (
      <div key={set.id}>
        <h3>Hero {set.id.toUpperCase()} ({set.topics}) – {variant}</h3>
        <IconGallery>
          {set.names.map((name) => (
            <IconItem key={name} name={name}>
